    const canvas = canvasRef.current
    const section = sectionRef.current
    if (!canvas || !section) return

    const reduced = prefersReducedMotion()

    let renderer: THREE.WebGLRenderer
    try {
      renderer = new THREE.WebGLRenderer({
        canvas, antialias: false, alpha: true, powerPreference: "high-performance",
      })
    } catch {
      // No WebGL: the page still works, it simply does not have the visual.
      setFailed(true)
      return
    }
    if (!renderer.getContext()) { setFailed(true); return }

    renderer.setClearColor(0x000000, 0)
    const dpr = Math.min(window.devicePixelRatio || 1, 2)
    renderer.setPixelRatio(dpr)

    const scene = new THREE.Scene()
    const camera = new THREE.PerspectiveCamera(52, 1, 0.1, 200)

    // One point per firm on the register. The count is the real number, not a
    // round one chosen for performance — it is the claim the page makes.
    const geometry = buildField(TOTALS.firms)
    const material = new THREE.ShaderMaterial({
      vertexShader: particleVertex,
      fragmentShader: particleFragment,
      transparent: true,
      depthWrite: false,
      blending: THREE.AdditiveBlending,
      uniforms: {
        uProgress: { value: 0 },
        uTime: { value: 0 },
        uSize: { value: 2.6 },
        uPixelRatio: { value: dpr },
        uColdColor: { value: new THREE.Color("#5b6472") },
        uBrandColor: { value: new THREE.Color("#215cd0") },
        uSignalColor: { value: new THREE.Color("#48c07a") },
      },
    })
    const points = new THREE.Points(geometry, material)
    scene.add(points)

    const coastGeo = new THREE.BufferGeometry()
    coastGeo.setAttribute(
      "position",
      new THREE.Float32BufferAttribute(coastlineSegments(), 3),
    )
    const coastMat = new THREE.ShaderMaterial({
      vertexShader: coastVertex,
      fragmentShader: coastFragment,
      transparent: true,
      depthWrite: false,
      uniforms: {
        uProgress: { value: 0 },
        uColor: { value: new THREE.Color("#8fb0ea") },
      },
    })
    const coast = new THREE.LineSegments(coastGeo, coastMat)
    scene.add(coast)

    const resize = () => {
      const w = canvas.clientWidth || window.innerWidth
      const h = canvas.clientHeight || window.innerHeight
      renderer.setSize(w, h, false)
      camera.aspect = w / h
      // On a narrow screen the map is taller than it is wide, so widen the
      // field of view rather than letting it crop.
      camera.fov = w / h < 0.9 ? 66 : 52
      camera.updateProjectionMatrix()
    }
    resize()

    // Progress is read from layout, never accumulated. Scrubbing backwards
    // has to give exactly the same frame as scrubbing forwards.
    const readProgress = () => {
      const r = section.getBoundingClientRect()
      const scrollable = r.height - window.innerHeight
      if (scrollable <= 0) return 0
      return Math.min(1, Math.max(0, -r.top / scrollable))
    }

    let frame = 0
    const clock = new THREE.Clock()

    const render = () => {
      const p = progressRef.current
      material.uniforms.uProgress.value = p
      coastMat.uniforms.uProgress.value = p
      material.uniforms.uTime.value = reduced ? 0 : clock.getElapsedTime()

      // Camera choreography: a long push in from the census cloud to the
      // map, then a slow drift down the country as the funnel narrows.
      const ease = p * p * (3 - 2 * p)
      camera.position.set(
        Math.sin(p * 0.9) * 3.4,
        THREE.MathUtils.lerp(3.5, -1.2, ease),
        THREE.MathUtils.lerp(34, 12.5, ease),
      )
      camera.lookAt(0, THREE.MathUtils.lerp(1.5, -0.8, ease), 0)
      points.rotation.y = Math.sin(p * 1.4) * 0.10
      coast.rotation.y = points.rotation.y

      renderer.render(scene, camera)
      frame = requestAnimationFrame(render)
    }

    const onScroll = () => {
      const p = readProgress()
      progressRef.current = p
      let i = 0
      for (let k = 0; k < ACTS.length; k++) if (p >= ACTS[k].at - 0.14) i = k
      setAct(i)
    }

    onScroll()
    render()
    window.addEventListener("scroll", onScroll, { passive: true })
    window.addEventListener("resize", resize)

    return () => {
      cancelAnimationFrame(frame)
      window.removeEventListener("scroll", onScroll)
      window.removeEventListener("resize", resize)
      geometry.dispose()
      material.dispose()
      coastGeo.dispose()
      coastMat.dispose()
      renderer.dispose()
    }
